# Apps from Debian for EVERY Navi desktop (personal and work).
# Apps from other sources, and the personal/work ones, are in scripts/apps.sh.

# Browsers: Debian's builds, security fixes come through apt
firefox-esr
chromium

# Viewers and editor
micro                         # terminal editor with normal shortcuts (Ctrl+S/C/V)
xclip                         # lets micro and the terminal use the system clipboard
ristretto                     # pictures
tumbler                       # thumbnails for Ristretto
zathura                       # PDFs
zathura-pdf-poppler           # zathura's PDF backend

# Terminal utilities
btop                          # system monitor
bat                           # cat with colors (Debian names the binary batcat)
fzf                           # fuzzy search: Ctrl+R searches shell history
ripgrep                       # rg: search inside files
fd-find                       # find files by name (Debian names the binary fdfind)

# yazi previews and archives
chafa                         # image previews inside Alacritty
poppler-utils
ffmpegthumbnailer
7zip
jq

# Security tools
aircrack-ng                   # Wi-Fi auditing (airmon-ng, airodump-ng, aireplay-ng)
ieee-data                     # vendor names for MAC addresses in airodump-ng
nmap                          # network scanner
